The tree hash used below (python3 /tmp/tree_hash_check.py <dir>) is a
small standalone script -- NOT part of tamperrun.py -- that sha256-hashes
every (relative path, file bytes) pair and every directory path in a tree,
sorted, so it independently proves the fixture directory's bytes did not
change. Its source:

    import hashlib, os, sys

    def tree_hash(root):
        h = hashlib.sha256()
        entries = []
        for dirpath, dirnames, filenames in os.walk(root, followlinks=False):
            dirnames.sort()
            rel_dir = os.path.relpath(dirpath, root).replace(os.sep, "/")
            entries.append(("D", rel_dir))
            for name in sorted(filenames):
                rel = os.path.relpath(os.path.join(dirpath, name), root).replace(os.sep, "/")
                with open(os.path.join(dirpath, name), "rb") as fh:
                    data = fh.read()
                entries.append(("F", rel, hashlib.sha256(data).hexdigest(), len(data)))
        for e in sorted(entries):
            h.update(repr(e).encode())
        return h.hexdigest()

    print(tree_hash(sys.argv[1]))
