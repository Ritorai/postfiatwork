    env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1',
               GIT_AUTHOR_NAME='evidence', GIT_AUTHOR_EMAIL='evidence@invalid',
               GIT_COMMITTER_NAME='evidence', GIT_COMMITTER_EMAIL='evidence@invalid',
               GIT_AUTHOR_DATE='2000-01-01T00:00:00Z',

    def git(*args):
        return subprocess.run(['git'] + list(args), cwd=dest, env=env,
                              stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    
    for argv in (['init', '-q', '-b', 'main'], ['add', '-A'],
                 ['commit', '-q', '-m', 'the tree as this deliverable commits it']):
        r = git(*argv)

            print(r.stdout.decode('utf-8', 'replace'))

    report = os.path.join(tool, 'self_scan_report.json')
    
    def sha(path):
        with open(path, 'rb') as fh:
            return hashlib.sha256(fh.read()).hexdigest()
