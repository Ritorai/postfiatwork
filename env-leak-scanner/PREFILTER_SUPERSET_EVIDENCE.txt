"""Count lines that fire a rule but that PREFILTER_RE rejects."""
import string
import sys

sys.path.insert(0, sys.argv[1])
import leakscan as L

CORES = [
    "/home/rito/out.json", "/Users/rito/out.json",
    "\\\\fileserv\\share\\x", "\\\\_fileserv\\share\\x",
    "\\\\.fileserv\\share\\x", "\\\\-fileserv\\share\\x",
]
AFFIXES = [""] + [c for c in string.printable if c not in "\n\r\x0b\x0c"]

checked = 0
missed = []
for core in CORES:
    for pre in AFFIXES:
        line = pre + core
        hits = L.scan_line("f.md", 1, line)
        if not hits:
            continue
        checked += 1
        if not L.PREFILTER_RE.search(line):
            missed.append((line, sorted({h["rule"] for h in hits})))

print("lines that fire a rule: %d" % checked)
print("of those, dropped by the prefilter: %d" % len(missed))
for line, rules in missed[:8]:
    print("  DROPPED %-40r %s" % (line, rules))
if len(missed) > 8:
    print("  ... and %d more" % (len(missed) - 8))
sys.exit(1 if missed else 0)
exit=0

--- 7. impact on this repository, measured per file ---
A full re-scan does move, and the honest way to report that is per file
rather than as one total. Outside env-leak-scanner/ NOT ONE finding
changes: the counterexamples do not occur in the repository's prior text,
so on the pre-existing corpus this fix is behaviour-preserving.

Inside env-leak-scanner/ the count does go up, because this commit's own
README section and this evidence file quote `/home/rito/...` and
`../home/alice/...` at the reader on purpose. That is limitation 4 --
"this directory will report itself" -- doing what it says. No exact
figure is quoted for those two files, and the reason is worth stating:
this file's content is part of what is being scanned, so any number
printed here would change the next time it is regenerated. A
self-referential count is not a fact about the repository.

"""Per-file confirmed-finding counts, before vs after, outside this dir."""
import collections
import json
import sys

SELF = "env-leak-scanner/"


def counts(path):
    with open(path, encoding="utf-8") as fh:
        rep = json.load(fh)
    return collections.Counter(f["file"] for f in rep["confirmed_leaks"])


before, after = counts(sys.argv[1]), counts(sys.argv[2])
outside = sorted({f for f in set(before) | set(after)
                  if not f.startswith(SELF)})
changed = [f for f in outside if before.get(f, 0) != after.get(f, 0)]
print("files outside %s with a changed count: %d" % (SELF, len(changed)))
for f in changed:
    print("  %-55s %3d -> %3d" % (f, before.get(f, 0), after.get(f, 0)))
tb = sum(before[f] for f in outside)
ta = sum(after[f] for f in outside)
print("confirmed findings outside %s: %d -> %d" % (SELF, tb, ta))
print("inside %s: count is self-referential, see the note above" % SELF)
sys.exit(0 if not changed else 1)
exit=0

--- 10. the reference tree was not written to ---

__pycache__, which is in .gitignore, so it would report a clean tree even
after something had imported a module inside it. This script also sets
PYTHONDONTWRITEBYTECODE=1 so the write does not happen in the first place.
